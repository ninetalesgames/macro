rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    function isOwner(userId) {
      return request.auth != null && request.auth.uid == userId;
    }

    function isValidEntry() {
      let data = request.resource.data;

      return data.keys().hasOnly([
          'calories',
          'protein',
          'weight',
          'notes',
          'updatedAt'
        ])
        && (!('calories' in data) || (data.calories is int && data.calories >= 0))
        && (!('protein' in data) || (data.protein is int && data.protein >= 0))
        && (!('weight' in data) || (data.weight is number && data.weight >= 0))
        && (!('notes' in data) || (data.notes is string && data.notes.size() <= 2000))
        && data.updatedAt is timestamp;
    }

    match /users/{userId}/entries/{dateKey} {
      allow read: if isOwner(userId);
      allow create, update: if isOwner(userId)
        && dateKey.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$')
        && isValidEntry();
      allow delete: if isOwner(userId);
    }
  }
}
